パターン1：モーダルダイアログ

ユーザーの操作を中断して確認を求める。出現は速く・閉じるは少し速めで。

// 表示
function openModal(overlay, box) {
  overlay.style.display = 'flex';
  anime({ targets: overlay, opacity: [0, 1], duration: 200, easing: 'linear' });
  anime({ targets: box, scale: [0.85, 1], opacity: [0, 1], duration: 350, easing: 'easeOutBack' });
}

// 閉じる
function closeModal(overlay, box) {
  anime({
    targets: [box, overlay],
    opacity: [1, 0],
    scale: { value: [1, 0.92], targets: box },
    duration: 250,
    easing: 'easeInCubic',
    complete: () => overlay.style.display = 'none'
  });
}

パターン2：トースト通知

画面端から素早く出現し、一定時間後に自動消去。ユーザーを待たせない速さが重要。

function showToast(message, duration = 3000) {
  const toast = document.createElement('div');
  toast.className = 'toast';
  toast.textContent = message;
  document.body.appendChild(toast);

  // 出現（easeOutExpoで速く・鋭く）
  anime({ targets: toast, translateX: [100, 0], opacity: [0, 1], duration: 400, easing: 'easeOutExpo' });

  // 自動消去
  setTimeout(() => {
    anime({
      targets: toast,
      translateX: [0, 100],
      opacity: [1, 0],
      duration: 300,
      easing: 'easeInCubic',
      complete: () => toast.remove()
    });
  }, duration);
}

パターン3：ドロワー（サイドメニュー）

モバイルナビゲーション等。translateXでDOM外から滑り込ませる。

let drawerOpen = false;

function toggleDrawer(drawer, overlay) {
  drawerOpen = !drawerOpen;
  if (drawerOpen) {
    overlay.style.display = 'block';
    anime([
      { targets: overlay, opacity: [0, 1], duration: 200, easing: 'linear' },
      { targets: drawer, translateX: ['-100%', '0%'], duration: 350, easing: 'easeOutCubic' }
    ]);
  } else {
    anime({
      targets: [drawer, overlay],
      translateX: { value: '-100%', targets: drawer },
      opacity: { value: 0, targets: overlay },
      duration: 280,
      easing: 'easeInCubic',
      complete: () => overlay.style.display = 'none'
    });
  }
  drawerOpen = !drawerOpen; // 正確なフラグ管理
}

パターン4：スケルトンUI（ローディング）

データ読み込み中のプレースホルダー。opacityをloopで脈動させる。

function startSkeletonPulse(skeletonEls) {
  return anime({
    targets: skeletonEls,
    opacity: [0.3, 1],
    easing: 'easeInOutSine',
    duration: 800,
    loop: true,
    direction: 'alternate',
    delay: anime.stagger(80)
  });
}

// データ取得完了後
async function loadAndRender(container) {
  const skeletons = container.querySelectorAll('.skeleton');
  const pulse = startSkeletonPulse(skeletons);

  const data = await fetchData();
  pulse.pause();
  renderContent(container, data);

  // コンテンツを滑らかに表示
  anime({ targets: container.querySelectorAll('.content-item'), opacity: [0, 1], translateY: [10, 0], delay: anime.stagger(60) });
}

パターン5：スクロール連動フェードイン

Intersection Observerと組み合わせて、要素が画面内に入ったときにアニメーション。

const reduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (!entry.isIntersecting) return;
    const el = entry.target;
    anime({
      targets: el,
      opacity: [0, 1],
      translateY: reduced ? 0 : [30, 0],
      duration: reduced ? 200 : 600,
      easing: 'easeOutCubic'
    });
    observer.unobserve(el);
  });
}, { threshold: 0.15 });

document.querySelectorAll('[data-animate]').forEach(el => {
  el.style.opacity = '0';
  observer.observe(el);
});

パターン6：タイムライン（LP ヒーロー入場）

ランディングページのファーストビュー。要素を「背景→見出し→説明→CTA」の順で演出。

anime.timeline({ easing: 'easeOutCubic' })
  .add({ targets: '.hero-bg', opacity: [0, 1], duration: 600 })
  .add({ targets: '.hero-headline', translateY: [30, 0], opacity: [0, 1], duration: 500 }, '-=200')
  .add({ targets: '.hero-desc', translateY: [20, 0], opacity: [0, 1], duration: 400 }, '-=200')
  .add({ targets: '.hero-cta', scale: [0.9, 1], opacity: [0, 1], duration: 350, easing: 'easeOutBack' }, '-=150');
